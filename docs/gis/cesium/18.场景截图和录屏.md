## 录屏

通过 Cesium 的 canvas 渲染画面实现视频录制，并导出为视频文件。

> [!NOTE] 关键步骤
>
> 1. 获取 Cesium 渲染的 canvas；
> 2. 使用 `captureStream()` 把 canvas 转为视频流；
> 3. 使用 `MediaRecorder` 录制视频流，并在录制过程中产生视频片段 Blob；
> 4. 停止录制时合并为完整视频并下载;

::: code-group

```ts [视频录制类] {16-20,42-48}
export default class GM_MapRecord {
  private mediaRecorder: MediaRecorder | null = null;
  private recordedChunks: Blob[] = [];

  constructor() {
    this.initRecord();
  }

  initRecord() {
    const canvasElt = document.querySelector(".cesium-widget canvas") as HTMLCanvasElement;
    // 捕获 canvas 流，并设置120FPS
    const mediaStream = canvasElt.captureStream(120);
    // 初始化媒体录制类，指定录制为 webm 格式视频
    this.mediaRecorder = new MediaRecorder(mediaStream, { mimeType: "video/webm;codecs=vp9" });
    // 当调用 mediaRecorder.stop() 时，才会异步触发，向 this.recordedChunks 添加数据流
    this.mediaRecorder.ondataavailable = (e) => {
      if (e.data.size > 0) {
        this.recordedChunks.push(e.data);
      }
    };
  }

  startRecord() {
    this.mediaRecorder?.start();
  }

  pauseRecord() {
    this.mediaRecorder?.pause();
  }

  resumeRecord() {
    this.mediaRecorder?.resume();
  }

  /**
   * 结束录制
   * @description ondataavailable是异步的，因此停止录制时，也需要通过异步才能获取到视频流
   */
  stopRecord(): Promise<Blob> {
    return new Promise((resolve) => {
      if (!this.mediaRecorder) return;
      this.mediaRecorder.onstop = () => {
        const blob = new Blob(this.recordedChunks, { type: "video/webm" });
        // 清空缓存，方便下次录制
        this.recordedChunks = [];
        // 把当前录制的视频流返回出去
        resolve(blob);
      };
      // 结束录制
      this.mediaRecorder.stop();
    });
  }
}
```

```vue [使用方式] {28,29}
<script setup lang="ts">
  import FileSaver from "file-saver";
  import { onMounted, ref } from "vue";
  import { createMap } from "@/utils/initEarth.ts";
  import { GM_MapRecord } from "@/helpers/GM_MapRecord.ts";

  let mapRecord: GM_MapRecord;

  onMounted(() => {
    createMap("cesium-container");
    mapRecord = new GM_MapRecord();
  });

  function startRecord() {
    mapRecord.startRecord();
  }

  function pauseRecord() {
    mapRecord.pauseRecord();
  }

  function resumeRecord() {
    mapRecord.resumeRecord();
  }

  async function stopRecord() {
    // 需要异步获取数据流
    const blob = await mapRecord.stopRecord();
    FileSaver.saveAs(blob, `屏幕录制_${Date.now()}.webm`);
  }
</script>
```
