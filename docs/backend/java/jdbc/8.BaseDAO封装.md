```java [通用增删改] {17}
public class BaseDAO {
  /**
   * 通用增删改方法
   * @param sql SQL语句
   * @param params 占位符赋值内容
   * @return 操作后受影响行数
   */
  public int executeUpdate(String sql, Object... params) throws SQLException {
    // 通过自己封装的 JDBCUtil 获取连接对象
    Connection connection = JDBCUtil.getConnection();
    // 初始化 preparedStatement
    PreparedStatement preparedStatement = connection.prepareStatement(sql);

    if (params != null && params.length > 0) {
      for (int i = 0; i < params.length; i++) {
        // 为占位符赋值时，索引是从 1 开始的
        preparedStatement.setObject(i + 1, params[i]);
      }
    }
    int result = preparedStatement.executeUpdate();

    // 释放资源
    preparedStatement.close();
    JDBCUtil.release();

    return result;
  }
}
```

```java [EmployeeDaoImpl]
public class EmployeeDaoImpl extends BaseDAO implements EmployeeDao {
  @Override
  public int insert(Employee employee) {
    try {
      String sql = "insert into t_emp (emp_name, emp_salary, emp_age) VALUES (?,?,?)";
      String empName = employee.getEmpName();
      Double empSalary = employee.getEmpSalary();
      Integer empAge = employee.getEmpAge();
      return executeUpdate(sql, empName, empSalary, empAge);
    } catch (SQLException e) {
      throw new RuntimeException(e);
    }
  }

  @Override
  public int update(Employee employee) {
    try {
      String sql = "update t_emp set emp_age = ? where emp_id = ?";
      Integer empId = employee.getEmpId();
      Integer empAge = employee.getEmpAge();
      return executeUpdate(sql, empAge, empId);
    } catch (SQLException e) {
      throw new RuntimeException(e);
    }
  }

  @Override
  public int delete(Integer empId) {
    try {
      String sql = "delete from t_emp where emp_id = ?";
      return executeUpdate(sql, empId);
    } catch (SQLException e) {
      throw new RuntimeException(e);
    }
  }
}
```

## 通用查询方法

::: code-group

```java [通用查询] {25,26,33,35,36,38,39}
public class BaseDAO {
  /**
   * 通用查询方法
   * @param clazz 结果类
   * @param sql SQL语句
   * @param params 查询参数
   * @return 查询到的集合
   * @param <T> 查询结果泛型类
   */
  public <T> List<T> executeQuery(Class<T> clazz, String sql, Object... params) throws Exception {
    // 获取连接对象
    Connection connection = JDBCUtil.getConnection();
    // 初始化 preparedStatement
    PreparedStatement preparedStatement = connection.prepareStatement(sql);

    if (params != null && params.length > 0) {
      for (int i = 0; i < params.length; i++) {
        // 为占位符赋值时，索引是从 1 开始的
        preparedStatement.setObject(i + 1, params[i]);
      }
    }
    ResultSet resultSet = preparedStatement.executeQuery();

    // 获取结果集中的元数据对象
    ResultSetMetaData metaData = resultSet.getMetaData();
    int columnCount = metaData.getColumnCount();

    List<T> list = new ArrayList<>();
    while (resultSet.next()) {
      T t = clazz.newInstance();
      // 每遍历一次，就带有有一行数据
      for (int i = 1; i <= columnCount; i++) {
        Object value = resultSet.getObject(i);
        // 获取到的 value 值，就是 t 这个对象中的一个属性
        String fieldName = metaData.getColumnLabel(i);
        Field field = clazz.getDeclaredField(fieldName);
        // 突破封装的 private
        field.setAccessible(true);
        field.set(t, value);
      }
      list.add(t);
    }

    // 释放资源
    resultSet.close();
    preparedStatement.close();
    JDBCUtil.release();

    return list;
  }
}
```

## 通用查询单个结果方法

::: code-group

```java [通用查询单个结果]
public class BaseDAO {
  /**
   * 通用查询单个结果的方法
   * @param clazz  结果类
   * @param sql    SQL语句
   * @param params 查询参数
   * @param <T>    查询结果泛型类
   * @return 查询到的集合
   */
  public <T> T executeQueryBean(Class<T> clazz, String sql, Object... params) throws Exception {
    List<T> list = this.executeQuery(clazz, sql, params);
    if (list == null || list.isEmpty()) {
      return null;
    }
    return list.get(0);
  }
}
```
