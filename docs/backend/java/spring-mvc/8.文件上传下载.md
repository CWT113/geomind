# 文件上传下载

在 Spring6 与 SpringMVC 环境下，文件上传推荐直接基于原生的 Servlet3.0 + Multipart 标准来实现。

## 引入依赖

在 `pom.xml` 中引入 `spring-webmvc` 以及 Jakarta Servlet API：

```xml
<dependencies>
  <!-- 引入 spring-webmvc 依赖 -->
  <dependency>
    <groupId>org.springframework</groupId>
    <artifactId>spring-webmvc</artifactId>
    <version>6.2.19</version>
  </dependency>
  <!-- 引入 servlet-api 依赖 -->
  <dependency>
    <groupId>jakarta.servlet</groupId>
    <artifactId>jakarta.servlet-api</artifactId>
    <version>6.1.0</version>
    <scope>provided</scope>
  </dependency>
</dependencies>
```



## 开启 Multipart 配置



::: code-group

```java [WebAppInitializer]
public class WebAppInitializer extends AbstractAnnotationConfigDispatcherServletInitializer {
  @Nullable
  @Override
  protected Class<?>[] getRootConfigClasses() {
    return null;
  }

  @Nullable
  @Override
  protected Class<?>[] getServletConfigClasses() {
    // 指向自定义的配置类 WebMvcConfig
    return new Class<?>[]{WebMvcConfig.class};
  }

  @Override
  protected String[] getServletMappings() {
    return new String[]{"/"};
  }

  /**
   * 等价于在 web.xml 的 <servlet> 中配置 <multipart-config>
   */
  @Override
  protected void customizeRegistration(ServletRegistration.Dynamic registration) {
    // 临时文件存储目录（null 或 "" 表示使用底层 Servlet 容器默认的临时工作区）
    String location = System.getProperty("java.io.tmpdir");
    // 对应 <max-file-size>10240000</max-file-size> (单个文件最大约 10MB)
    long maxFileSize = 10240000L;
    // 对应 <max-request-size>102400000</max-request-size> (整个请求最大约 100MB)
    long maxRequestSize = 102400000L;
    // 对应 <file-size-threshold>0</file-size-threshold>
    int fileSizeThreshold = 0;
    // 实例化 MultipartConfigElement
    MultipartConfigElement multipartConfig = new MultipartConfigElement(
      location,
      maxFileSize,
      maxRequestSize,
      fileSizeThreshold
    );
    registration.setMultipartConfig(multipartConfig);
  }
}
```

@EnableWebMvc
@ComponentScan("com.example")
public class WebMvcConfig implements WebMvcConfigurer {
  @Override
  public void extendMessageConverters(List<HttpMessageConverter<?>> converters) {
    // 允许使用 Jackson 把 HTTP 请求里的 JSON 转换为 Java 对象

## 文件上传

```java
@RestController
@RequestMapping("/file")
public class FileController {
  // 存储目录
  private static final String UPLOAD_DIR = System.getProperty("user.home") + "/uploads";

  @PostMapping("/upload")
  public ResponseEntity<String> uploadFile(@RequestParam("file") MultipartFile file) {
    if (file.isEmpty()) {
      return ResponseEntity.badRequest().body("上传失败，文件为空！");
    }
    try {
      File destDir = new File(UPLOAD_DIR);
      if (!destDir.exists()) {
        destDir.mkdirs();
      }
      // 防止文件重名，保存携带 UUID 前缀
      String originalFilename = file.getOriginalFilename();
      String safeFilename = UUID.randomUUID() + "_" + originalFilename;
      Path targetPath = Paths.get(UPLOAD_DIR, safeFilename);
      // 保存到磁盘
      file.transferTo(targetPath.toFile());

      return ResponseEntity.ok("上传成功，存储文件名：" + safeFilename);
    } catch (Exception e) {
      return ResponseEntity.internalServerError().body("上传异常：" + e.getMessage());
    }
  }
}
```



## 文件下载

```java
@RestController
@RequestMapping("/file")
public class FileController {
  // 存储目录
  private static final String UPLOAD_DIR = System.getProperty("user.home") + "/uploads";

  @GetMapping("/download")
  public ResponseEntity<Resource> download(@RequestParam("filename") String filename) {
    try {
      // 防止路径穿越攻击，只保留纯文件名
      String cleanName = Paths.get(filename).getFileName().toString();
      // 将上传根目录与过滤后的纯文件名安全拼接，并调用 normalize() 清理掉路径中冗余的路径符号
      Path filePath = Paths.get(UPLOAD_DIR).resolve(cleanName).normalize();
      File file = filePath.toFile();

      // 校验文件是否存在，以及当前程序是否有读取该文件的权限
      if (!file.exists() || !file.canRead()) {
        return ResponseEntity.notFound().build();
      }
      // 将 File 包装为 Spring 提供的 FileSystemResource 对象，支持 Spring 框架内部流式传输和零拷贝优化
      Resource resource = new FileSystemResource(file);
      // 根据文件后缀和内容获取其 MIME 类型
      String contentType = Files.probeContentType(filePath);
      if (contentType == null) {
        // 如果没有获取到 MIME 类型，则设置为兜底的通用二进制数据流类型
        contentType = MediaType.APPLICATION_OCTET_STREAM_VALUE;
      }
      // 处理中文文件名乱码
      String encodedFilename = URLEncoder.encode(cleanName, StandardCharsets.UTF_8).replaceAll("\\+", "%20");

      return ResponseEntity.ok()
        // 设置 Content-Type 响应头
        .contentType(MediaType.parseMediaType(contentType))
        // 设置 Content-Disposition 响应头，触发下载而非浏览器内联打开
        .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename*=UTF-8''" + encodedFilename)
        // 设置 Content-Length 响应头，通知客户端文件实际字节大小
        .contentLength(file.length())
        // 将文件流资源写入响应体，出发下载传输
        .body(resource);
    } catch (Exception e) {
      return ResponseEntity.internalServerError().build();
    }
  }
}
```
