# 阻塞队列

阻塞队列（Blocking Queue）本质上是 **带有线程同步机制的队列**，在并发环境中用来安全的传递数据。

> [!NOTE] 核心过程
>
> - **队列为空时**：消费者线程尝试获取元素会被挂起，直到有生产者放入元素；
> - **队列已满时**：生产者线程尝试放入元素会被挂起，直到有消费者取走元素；

通过这两点，它自然的实现了 **生产者-消费者模型**，不用自己写复杂的锁和条件变量。



## 单锁实现

```java [BlockingQueue1]
public class BlockingQueue1<E> implements BlockingQueue<E> {
  private final E[] array;
  private int head = 0;
  private int tail = 0;
  private int size = 0;

  private final ReentrantLock lock = new ReentrantLock(); // 锁
  private final Condition headWaits = lock.newCondition(); // 模拟消费者阻塞
  private final Condition tailWaits = lock.newCondition(); // 模拟生产者阻塞

  @SuppressWarnings("all")
  public BlockingQueue1(int capacity) {
    array = (E[]) new Object[capacity];
  }

  @Override
  public void offer(E e) throws InterruptedException {
    lock.lockInterruptibly();
    try {
      // 判断队列是否已满，满了就阻塞线程，同时用循环判断来避免虚拟唤醒
      while (isFull()) {
        tailWaits.await(); // 阻塞线程
      }
      array[tail] = e;
      // tail指针归 0，避免索引越界
      if (++tail == array.length) {
        tail = 0;
      }
      size++;
      // 唤醒 poll 中阻塞的线程，表示有数据可以取出了
      headWaits.signal();
    } finally {
      lock.unlock();
    }
  }

  @Override
  public boolean offer(E e, long timeout) throws InterruptedException {
    lock.lockInterruptibly();
    try {
      // 毫秒转为纳秒
      long t = TimeUnit.MILLISECONDS.toNanos(timeout);
      while (isFull()) {
        // 表示阻塞的线程倒计时到了，自动退出阻塞
        if (t <= 0) {
          return false;
        }
        // 返回值代表的是剩余时间
        t = tailWaits.awaitNanos(t);
      }
      array[tail] = e;
      if (++tail == array.length) {
        tail = 0;
      }
      size++;
      // 唤醒 poll 中阻塞的线程，表示有数据可以取出了
      headWaits.signal();
      return true;
    } finally {
      lock.unlock();
    }
  }

  @Override
  public E poll() throws InterruptedException {
    lock.lockInterruptibly();
    try {
      // 判断队列是否为空，为空就阻塞线程，同时用循环判断来避免虚拟唤醒
      while (isEmpty()) {
        headWaits.await();
      }
      E e = array[head];
      if (++head == array.length) {
        head = 0;
      }
      size--;
      // 唤醒 offer 中阻塞的队列，表示有数据可以入队了
      tailWaits.signal();
      return e;
    } finally {
      lock.unlock();
    }
  }

  public boolean isFull() {
    return size == array.length;
  }

  public boolean isEmpty() {
    return size == 0;
  }

  @Override
  public String toString() {
    return Arrays.toString(array);
  }
}
```

```java [BlockingQueue]
public interface BlockingQueue<E> {
  /**
	 * 元素入队
	 * @param e 元素值
	 * @throws InterruptedException 阻塞异常
	 */
  void offer(E e) throws InterruptedException;

  /**
	 * 元素入队（带延时）
	 * @param e       元素值
	 * @param timeout 超时时间（超过该时间，线程自动解锁）
	 * @return true：入队成功；false：入队失败
	 * @throws InterruptedException 阻塞异常
	 */
  boolean offer(E e, long timeout) throws InterruptedException;

  /**
	 * 元素出队
	 * @return 元素值
	 * @throws InterruptedException 阻塞异常
	 */
  E poll() throws InterruptedException;
}
```

```java [单元测试]
// 注意：不要再 JUnit 单元测试中测试，单元测试也是一个进程！
public static void main(String[] args) throws InterruptedException {
  BlockingQueue1<String> queue = new BlockingQueue1<>(3);

  Thread t1 = new Thread(() -> {
    try {
      System.out.println("入队前：" + System.currentTimeMillis());

      queue.offer("任务1");
      System.out.println(queue);
      queue.offer("任务2");
      System.out.println(queue);
      queue.offer("任务3");
      System.out.println(queue);

      // 入队失败，因为队列已经满了
      queue.offer("任务4", 5000);
      System.out.println(queue);

      System.out.println("入队后：" + System.currentTimeMillis());
    } catch (InterruptedException e) {
      throw new RuntimeException(e);
    }
  }, "生产者");
  t1.start();

  Thread.sleep(2000);
  // 2秒钟以后，取一个数据，任务4就能加入队列了
  queue.poll();
}
```

## 双锁实现

```java {32-35,44-51,74,76-78,84-91}
public class BlockingQueue2<E> implements BlockingQueue<E> {
  private final E[] array;
  private int head = 0;
  private int tail = 0;
  // 使用原子变量，避免多线程中 int 类型自增自减错乱
  private final AtomicInteger size = new AtomicInteger();

  private final ReentrantLock headLock = new ReentrantLock(); // 头部出队锁
  private final Condition headWaits = headLock.newCondition();

  private final ReentrantLock tailLock = new ReentrantLock(); // 尾部入队锁
  private final Condition tailWaits = tailLock.newCondition();

  @SuppressWarnings("all")
  public BlockingQueue2(int capacity) {
    array = (E[]) new Object[capacity];
  }

  @Override
  public void offer(E e) throws InterruptedException {
    tailLock.lockInterruptibly();
    int c;
    try {
      while (isFull()) {
        tailWaits.await();
      }
      array[tail] = e;
      if (++tail == array.length) {
        tail = 0;
      }
      // 使用原子变量的自增方法，返回值是自增前的值
      c = size.getAndIncrement();
      if (c + 1 < array.length) {
        tailWaits.signal();
      }
    } finally {
      tailLock.unlock();
    }

    // 关键点：
    // 1.唤醒出队中阻塞的线程，必须要和 headLock 配对使用；
    // 2.写在 tailLock 加锁和解锁外面，避免死锁问题；
    // 3.使用 size 自增前的值，判断是否是第一次唤醒锁；
    if (c == 0) {
      headLock.lock();
      try {
        headWaits.signal();
      } finally {
        headLock.unlock();
      }
    }
  }

  @Override
  public boolean offer(E e, long timeout) throws InterruptedException {
    return false;
  }

  @Override
  public E poll() throws InterruptedException {
    E e;
    int c;
    headLock.lockInterruptibly();
    try {
      while (isEmpty()) {
        headWaits.await();
      }
      e = array[head];
      array[head] = null; // help GC
      if (++head == array.length) {
        head = 0;
      }
      // 获取减少前的 size 值
      c = size.getAndDecrement();
      // 如果队列不是空的，级联唤醒下一个线程
      if (c > 1) {
        headWaits.signal();
      }
    } finally {
      headLock.unlock();
    }

    // 唤醒入队中阻塞的线程，同样必须要配对使用
    if (c == array.length) {
      tailLock.lock();
      try {
        tailWaits.signal();
      } finally {
        tailLock.unlock();
      }
    }

    return e;
  }

  @Override
  public String toString() {
    return Arrays.toString(array);
  }

  public boolean isFull() {
    return size.get() == array.length;
  }

  public boolean isEmpty() {
    return size.get() == 0;
  }
}
```
