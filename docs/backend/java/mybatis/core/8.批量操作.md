# 批量操作



## 批量新增

批量新增有多种方式实现，如 **`foreach` 拼接批量 `insert`（推荐）**、JDBC 批处理执行、循环单条 `insert` 等。

::: code-group

```xml {11-13} [SysUserMapper.xml]
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper
        PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.example.mapper.SysUserMapper">

  <insert id="batchInsert">
    insert into sys_user
      (user_id, user_name, nick_name, password, email, create_time, update_time)
    values
      <foreach collection="sysUsers" item="u" separator=",">
        (null, #{u.userName}, #{u.nickName}, #{u.password}, #{u.email}, now(), now())
      </foreach>
  </insert>
  
</mapper>
```

```java [SysUserMapper]
public interface SysUserMapper {
  int batchInsert(@Param("sysUsers") List<SysUser> sysUsers);
}
```

```java [单元测试]
public class AppTest {
  @Test
  void testBatchInsert() {
    SqlSession sqlSession = MybatisUtils.getSqlSession(true);
    SysUserMapper mapper = sqlSession.getMapper(SysUserMapper.class);
    List<SysUser> sysUsers = new ArrayList<>();
    for (int i = 0; i < 10; i++) {
      SysUser user = SysUser.builder()
        .userName("user" + i)
        .nickName("nick" + i)
        .password("password" + i)
        .email("email" + i)
        .build();
      sysUsers.add(user);
    }
    int affectRows = mapper.batchInsert(sysUsers);
    System.out.println(affectRows);
  }
}
```

## 批量更新

::: code-group

```xml [SysUserMapper.xml]
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper
        PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.example.mapper.SysUserMapper">

	<update id="batchUpdate">
    update
      sys_user
    set
      user_name = case user_id
        <foreach collection="sysUsers" item="u">
          when #{u.userId} then #{u.userName}
        </foreach>
      end,
      nick_name = case user_id
        <foreach collection="sysUsers" item="u">
          when #{u.userId} then #{u.nickName}
        </foreach>
      end
    where user_id in
      <foreach collection="sysUsers" item="u" open="(" separator="," close=")">
        #{u.userId}
      </foreach>
  </update>
  
</mapper>
```

```java [SysUserMapper]
public interface SysUserMapper {
  int batchUpdate(@Param("sysUsers") List<SysUser> sysUsers);
}
```

```java [单元测试]
public class AppTest {
  @Test
  void testBatchUpdate() {
    SqlSession sqlSession = MybatisUtils.getSqlSession(true);
    SysUserMapper mapper = sqlSession.getMapper(SysUserMapper.class);
    List<SysUser> sysUsers = new ArrayList<>();
    for (int i = 0; i < 10; i++) {
      SysUser user = SysUser.builder()
        .userId(i + 1)
        .userName("admin" + i)
        .nickName("超级管理员" + i)
        .build();
      sysUsers.add(user);
    }
    int affectRows = mapper.batchUpdate(sysUsers);
    System.out.println(affectRows);
  }
}
```

## 批量删除

::: code-group

```xml {11-13} [SysUserMapper.xml]
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper
        PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.example.mapper.SysUserMapper">

  <delete id="batchDelete">
    delete from
      sys_user
    where user_id in
      <foreach collection="userIds" item="userId" open="(" separator="," close=")">
        #{userId}
      </foreach>
  </delete>
  
</mapper>
```

```java [SysUserMapper]
public interface SysUserMapper {
  int batchDelete(@Param("userIds") List<Integer> userIds);
}
```

```java [单元测试]
public class AppTest {
  @Test
  void testBatchDelete() {
    SqlSession sqlSession = MybatisUtils.getSqlSession(true);
    SysUserMapper mapper = sqlSession.getMapper(SysUserMapper.class);
    List<Integer> userIds = List.of(1, 2, 3, 4, 5);
    int affectRows = mapper.batchDelete(userIds);
    System.out.println(affectRows);
  }
}
```
