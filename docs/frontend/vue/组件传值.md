# 组件传值

## 父向子传值

> [!CAUTION] 注意
>
> 父向子传值必须要遵守 **“单向数据流”** 原则，子组件中不可更改 props 的值！



### defineProps

`defineProps` 是一个专门用来接收父组件传递过来的属性（Props）的编译器宏。

::: code-group

``` vue [Parent.vue]
<template>
  <child :name="name" :user="user"/>
</template>

<script setup lang="ts">
  import { reactive, ref } from "vue";
  import Child from "@/components/Child.vue";

  const name = ref("sunny");
  const user = reactive({
    age: 20,
    hobbies: ["篮球", "足球"],
  });
</script>
```

``` vue [Child.vue]
<template>
  <!-- 模板中不需要 props.name，会自动解构 -->
  <h3>姓名：{{ name }}</h3>
  <h3>年龄：{{ user?.age }}</h3>
  <h3>爱好：{{ user?.hobbies }}</h3>
</template>

<script setup lang="ts">
  interface Props {
    name?: string;
    user?: {
      age: number;
      hobbies: string[];
  	};
  }

  // 方式一：解构赋值方式设置默认值（✨V3.5+版本推荐方式）
  const { name = "默认名称", user = { age: 0, hobbies: ["无"] } } = defineProps<Props>(); // [!code ++]

  // 方式二：withDefaults宏设置默认值（注意：不要使用解构，值类型还是会失去响应式！）
  const props = withDefaults(defineProps<Props>(), {
    name: "默认名称",
    // 对象或数组类型的默认值必须使用工厂函数
    user: () => ({
      age: 0,
      hobbies: ["无"],
    }),
  });
</script>
```

:::



### defineExpose

这种方式属于 “父组件主动去拿”，而不是“子组件被动接收”。

::: code-group

``` vue [Parent.vue] {11}
<template>
  <button @click="updateName">更新子组件name值</button>

  <child ref="childRef"></child>
</template>

<script setup lang="ts">
  import { useTemplateRef } from "vue";
  import Child from "@/components/Child.vue";

  const childRef = useTemplateRef("childRef");

  function updateName() {
    childRef.value?.setName("王一博");
  }
</script>
```

``` vue [Child.vue] {14}
<template>
  <p>{{ name }}</p>
</template>

<script setup lang='ts'>
  import { ref } from "vue";

  const name = ref("");

  function setName(val: string) {
    name.value = val;
  }

  defineExpose({ setName });
</script>
```

:::




## 子向父传值

### emit

子组件通过 `defineEmits` 触发一个自定义事件，并附带需要传递的数据，然后父组件在模板中监听该事件。

::: code-group

``` vue [Parent.vue]
<template>
  <p>{{ childName }}</p>

  <child @update:name="handleUpdateName"></child>
</template>

<script setup lang="ts">
  import { ref } from "vue";
  import Child from "@/components/Child.vue";

  const childName = ref<string>("");

  function handleUpdateName(value: string) {
    childName.value = value;
  }
</script>
```

``` vue [Child.vue]
<template>
  <p>{{ name }}</p>

  <button @click="updateName">子组件更新 name 值</button>
</template>

<script setup lang='ts'>
  import { ref } from "vue";

  interface Emits {
    (e: "update:name", value: string): void;
  }

  const emit = defineEmits<Emits>();

  const name = ref("tom");

  function updateName() {
    name.value += "~";
    emit("update:name", name.value);
  }
</script>

```

:::



### defineExpose

> [!TIP] 提示
>
> 使用 `defineExpose` 向外暴漏属性时，推荐使用 `readonly` 修饰符进行包裹，这样可以有效避免外部更改组件内的值！

::: code-group

``` vue [Parent.vue]
<template>
  <child ref="childRef"></child>

  <button @click="updateName">更新子组件 name 值</button>
</template>

<script setup lang="ts">
  import { useTemplateRef } from "vue";
  import Child from "@/components/Child.vue";

  const childRef = useTemplateRef("childRef");

  function updateName() {
    // 错误做法：[Vue warn] Set operation on key "value" failed: target is readonly.
    childRef.value.name = "Alice"; // [!code error]

    // 正确做法
    childRef.value?.updateName("Alice"); // [!code ++]
  }
</script>
```

``` vue [Child.vue] {18,20-22}
<template>
	<p>{{ name }}</p>
</template>

<script setup lang='ts'>
  import { readonly, ref } from "vue";

  const name = ref("tom");

  function updateName(value: string) {
    name.value = value;
  }

  defineExpose({
    // 方式一：向外暴漏子组件属性，使用 readonly（vue3.3+）进行包装，避免外部更改其属性值
    name: readonly(name), // [!code ++]
    // 方式二：使用 computed 返回只读的属性
    name: computed(() => name.value),
    // 方式三：使用 getter 函数，只允许get，不允许设置set
    get name() {
      return name.value;
    },
    updateName,
  });
</script>
```
:::



### defineModel

`defineModel` 宏可以用于声明双向绑定的属性，父组件通过 `v-model` 使用，子组件通过 `defineModel` 接收。

::: code-group

```vue [Parent.vue] {2}
<template>
  <child v-model="name" v-model:count="count"/>

  <h2>{{ name }}</h2>
  <h2>{{ count }}</h2>
</template>

<script setup lang="ts">
  import { ref } from "vue";
  import Child from "@/components/Child.vue";

  const count = ref(0);
  const name = ref("sunny");
</script>
```

```vue [Child.vue] {8,11}
<template>
  <input v-model="name"/>
  <input v-model="count"/>
</template>

<script setup lang="ts">
  // 匿名model，父组件通过 v-model 使用
  const name = defineModel({ type: String });

  // 具名model，父组件通过 v-model:count 使用
  const count = defineModel("count", { type: Number, required: true });
</script>
```

:::



### 传递回调函数

父组件把一个函数作为 Props 传递给子组件，子组件要向父组件传值时，直接调用该函数并将数据当作参数传进去。

这种做法在 React 中很常见，在 Vue 中也同样适用。

::: code-group

```vue [Parent.vue] {2}
<template>
  <child :onCallback="receiveData"/>

  <h2>{{ name }}</h2>
</template>

<script setup lang="ts">
  import { ref } from "vue";
  import Child from "@/components/Child.vue";

  const name = ref("hello");

  function receiveData(data: any) {
    name.value = data;
  }
</script>

```

```vue [Child.vue] {6-8}
<template>
  <button @click="handleClick">向父组件传值</button>
</template>

<script setup lang="ts">
  const { onCallback } = defineProps({
    onCallback: Function,
  });

  function handleClick() {
    onCallback && onCallback("打卡成功！");
  }
</script>
```

:::




## 多层组件传值

### provide + inject

> [!CAUTION] 缺点
>
> - 只能在 `.vue` 的组件树中使用，不适合做全局状态管理；
>
> - 难以追踪数据来源，大型项目中状态管理变的很困难，维护成本增加；
>
> - 默认提供的值是静态的，如果提供非响应式的对象，后代组件无法感应变化，必须像如下这样声明：
>
>   ```ts
>   provide('state', reactive({ count: 0 }))
>   ```

::: code-group
``` vue [App.vue]
<template>
  <router-view/>
</template>

<script setup lang="ts">
  import { RouterView } from "vue-router";
  import { provide, reactive } from "vue";
  import { USER_INFO_KEY } from "@/types/injection-keys.ts";

  const userInfo = reactive({
    name: "",
    age: 25,
  });

  // App组件注入属性
  provide(USER_INFO_KEY, userInfo);
</script>
```
``` vue [Child.vue]
<template>
  <p>{{ userInfo.name }}</p>
  <p>{{ userInfo.age }}</p>
</template>

<script setup lang='ts'>
  import { inject } from "vue";
  import { USER_INFO_KEY } from "@/types/injection-keys.ts";

  // 注意：第二个参数表示 defaultValue，它只在未提供 provide 时才生效！
  const userInfo = inject(USER_INFO_KEY, {
    name: "tom",
    age: 1,
  });
</script>
```
``` typescript [injection-keys.ts]
import type { InjectionKey } from "vue";

export interface UserInfo {
  name?: string;
  age?: number;
}

export const USER_INFO_KEY: InjectionKey<UserInfo> = Symbol("USER_INFO");
```
:::



### eventBus

> [!TIP] 提示
> 可以了解使用 [mitt](https://github.com/developit/mitt) 第三方库，功能更全面。
>
> ``` bash
> pnpm install mitt --save
> ```

::: code-group
``` typescript [eventBus.ts]
type callback = (...args: any[]) => void;

class EventBus {
  private events = new Map<string, Set<callback>>();

  emit(event: string, ...args: any[]) {
    this.events.get(event)?.forEach((cb) => cb(...args));
  }

  on(event: string, cb: callback) {
    if (!this.events.has(event)) {
      this.events.set(event, new Set());
    }
    this.events.get(event)!.add(cb);
  }

  off(event: string) {
    this.events.delete(event);
  }

  once(event: string, cb: callback) {
    const wrapper = (...args: any[]) => {
      cb(...args);
      this.off(event);
    };
    this.on(event, wrapper);
  }
}

export const emitter = new EventBus();
```

``` vue [App.vue]
<template>
	<button @click="sendMsg">发送消息</button>
</template>

<script setup lang="ts">
  import { emitter } from "@/utils/eventBus.ts";

  function sendMsg() {
    // 传递单个字符串
    emitter.emit("SEND_MESSAGE", "hello,world");
    // 传递数组
    emitter.emit("SEND_MESSAGE1", [1, 2, 3]);
    // 传递对象
    emitter.emit("SEND_MESSAGE2", { name: "sunny" });
    // 只发布一次
    emitter.emit("SEND_MESSAGE3", { age: 25 });
  }
</script>
```

``` vue [Child.vue]
<script setup lang='ts'>
  import { onMounted } from "vue";
  import { emitter } from "@/utils/eventBus.ts";

  onMounted(() => {
    emitter.on("SEND_MESSAGE", (msg) => {
      console.log(msg);
    });

    emitter.on("SEND_MESSAGE1", (msg) => {
      console.log(msg);
    });

    emitter.on("SEND_MESSAGE2", (msg) => {
      console.log(msg);
    });

    // 只订阅一次
    emitter.once("SEND_MESSAGE3", (msg) => {
      console.log(msg);
    });

    setTimeout(() => {
      // 取消订阅
      emitter.off("SEND_MESSAGE2");
    }, 5000);
  });
</script>
```
