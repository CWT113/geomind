# 1.定义Store

``` vue
import { defineStore } from "pinia";
import { reactive, readonly, ref } from "vue";

export const useAppStore = defineStore("app", () => {
  const name = ref("sunny");
  const info = reactive({
    age: 25,
    hobbies: ["篮球", "足球"],
  });

  function setName(val: string) {
    name.value = val;
  }

  function setInfo(val: Partial<typeof info>) {
    if (!val || Object.keys(val).length == 0) {
      return;
    }
    Object.assign(info, val);
  }

  return {
    name: readonly(name),
    info: readonly(info),
    setName,
    setInfo,
  };
});
```
